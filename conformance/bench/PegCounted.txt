# peg-hs の記法（ADR 0004、0005、0012、0014）を peg-hs 自身で書いた文法。
# 生成したパーサーは最上位の要素の列を返し、組み立て（Peg.Grammar.Assemble）を通る。
# 誤りの文言は、ラベル（ADR 0021）から Peg.Grammar.Parser の対応表で作る（ADR 0027）。
# ラベルは、そこで失敗したらどの選択肢でも成功しえない位置にだけ付ける。
#
# 識別子と空白の文字は Peg.Grammar.Assemble の関数で判定する（ADR 0020）。
%module Conformance.Bench.PegCounted

# 次のラベルは、ラベルの安全性の検査（ADR 0028）が危険と判定するが、安全である。
# 識別子と空白の文字は意味述語で判定しており、検査は意味述語がどの文字でも成功
# しうるとみなすため、空白や識別子で始まる別の選択肢と区別できない。安全なことは、
# ラベルを外した文法との受理の一致のテストで確かめている。
%safe verbatim
%safe ident
%safe type
%safe argument
%safe paramform
%safe paramtype
%safe argcode
%safe expression
%safe groupaction
%safe paren
%safe literal
%safe escape
%safe ubrace
%safe uclose
%safe uescape
%safe emptyclass
%safe class
%safe range
%safe action
%safe hsstring
%safe hschar
%safe constructor

# 入れ子の深さの上限のラベル（ADR 0080）は、上限を超える文法を拒むためにある。
# 検査は、深さの意味述語が失敗した後に別の選択肢が成功しうるとみなすが、
# 上限の内側の文法の受理は変えない。
%safe nest

%{
import Data.Char (chr, digitToInt, isHexDigit, isSpace)
import Data.Text (Text)
import qualified Data.Text as T
import Peg.Grammar
import Peg.Grammar.Assemble
import Peg.Runtime (Pos (..), Span (..))

here :: Span -> Loc
here s = Loc (posLine (spanStart s)) (posColumn (spanStart s))

firstChar :: Text -> Char
firstChar = T.foldr const ' '

-- Characters of the inside of a literal or class, with escapes decoded.
-- The flag marks an unescaped '-', which can form a range in a class.
tokens :: Text -> [(Char, Bool)]
tokens = go . T.unpack
  where
    -- Recovery (ADR 0035) can reach here with an invalid escape.
    go ('\\' : 'u' : '{' : rest) =
      let (h, rest') = span isHexDigit rest
          n = foldl (\m d -> m * 16 + toInteger (digitToInt d)) 0 h
       in (if n <= 0x10FFFF then chr (fromInteger n) else '\xFFFD', False) : go (drop 1 rest')
    go ('\\' : c : rest) = (escape c, False) : go rest
    go (c : rest) = (c, c == '-') : go rest
    go [] = []
    escape c = case c of
      'n' -> '\n'
      't' -> '\t'
      'r' -> '\r'
      _ -> c

inside :: Text -> Text
inside = T.drop 1 . T.dropEnd 1

decodeLiteral :: Text -> Text
decodeLiteral = T.pack . map fst . tokens . inside

classRanges :: Text -> [(Char, Char)]
classRanges = go . tokens . inside
  where
    go ((a, _) : (_, True) : (b, _) : rest) = (a, b) : go rest
    go ((a, _) : rest) = (a, a) : go rest
    go [] = []

validHex :: Text -> Bool
validHex h = not (T.null h) && T.foldl' (\n d -> n * 16 + toInteger (digitToInt d)) 0 h <= (0x10FFFF :: Integer)

-- Every range of a class item is non-empty.
nonEmptyRanges :: Text -> Bool
nonEmptyRanges item = all (uncurry (<=)) (classRanges (T.concat [T.pack "[", item, T.pack "]"]))

paramFormMessage :: Text
paramFormMessage = T.pack "expected a parameter of the form name :: Type"

paramError :: Loc -> Text -> Maybe Text
paramError l c = either Just (const Nothing) (toParam (Located l c))

param :: Loc -> Text -> Param
param l c = either (const (Param T.empty (Located l c))) id (toParam (Located l c))

-- The limit on the nesting of expressions (ADR 0080).
maxDepth :: Int
maxDepth = 256

-- A construct one level deeper than what it contains.
deeper :: (a -> Expr) -> (a, Int) -> (Expr, Int)
deeper f (x, n) = (f x, n + 1)

-- The items of the sequences of a group, and the levels of the deepest.
withDepth :: [[(Item, Int)]] -> ([[Item]], Int)
withDepth ss = (map (map fst) ss, maximum (0 : map snd (concat ss)))

applyOp :: Expr -> Char -> Expr
applyOp e c = case c of
  '*' -> EStar e
  '+' -> EPlus e
  _ -> EOpt e
}%

File :: [Top]
File <- Ws TopItem* (!.) ^top

TopItem :: Top
TopItem <- TopOne Ws

TopOne :: Top
TopOne <- Verbatim
       / Directive
       / Declaration

Here :: Loc
Here <- { here _span }

# 空白とコメント -------------------------------------------------------------------
Ws <- (SpaceChar / Comment)*

SpaceChar <- c:$. &{ isSpace (firstChar c) } { () }

Comment <- '#' (!'\n' .)*

InlineSp <- [ \t]*

Ident :: Text
Ident <- $(IdentStart IdentChar*)

IdentStart <- c:$. &{ isIdentStart (firstChar c) } { () }

IdentChar <- c:$. &{ isIdentChar (firstChar c) } { () }

# 最上位の要素 ---------------------------------------------------------------------
Verbatim :: Top
Verbatim <- '%{' l:Here b:$(!'}%' .)* ('}%' ^verbatim) { uncurry TVerbatim (verbatimBody l b) }

Directive :: Top
Directive <- l:Here '%' n:(Ident ^ident) InlineSp al:Here raw:$(!'\n' .)* { TDirective l n al (directiveArg n raw) }

Declaration :: Top
# 三つめの選択肢は必ずラベルを送出する。最上位の識別子は宣言の始まりでしかありえない。
# 値は、回復（ADR 0035）したときにだけ使われる
Declaration <- l:Here n:Ident tps:TemplateParams? ps:ParamList? Ws '::' InlineSp tl:Here ty:(TypeText ^type) { TAnnot l n (maybe [] id ps) (maybe [] id tps) (Located tl ty) }
            / l:Here n:Ident tps:TemplateParams? Ws '<-' as:Alts { TDef l n (maybe [] (map templateParamName) tps) as }
            / Ident TemplateParams? ParamList? Ws (!'<-') ^params (('::' / '<-') ^decl) { TVerbatim (Loc 0 0) T.empty }

# 雛形の引数（ADR 0031）。型注釈では値を使う引数に型変数を付ける
TemplateParams :: [TemplateParam]
TemplateParams <- '<' Ws p:TemplateParam ps:(Ws ',' Ws TemplateParam)* Ws '>' { p : ps }

TemplateParam :: TemplateParam
TemplateParam <- n:Ident ty:(Ws '::' Ws Ident)? { TemplateParam n ty }

TypeText :: Text
TypeText <- ty:$(!'\n' .)* &{ not (T.null (T.stripEnd ty)) } { T.stripEnd ty }

ParamList :: [Param]
ParamList <- '(' ParamChunks

ParamChunks :: [Param]
ParamChunks <- p:ParamChunk ',' rest:ParamChunks { p : rest }
            / p:ParamChunk ')' { [p] }

ParamChunk :: Param
ParamChunk <- Ws l:Here c:$ArgCode (&{ not (T.null (T.strip c)) }) ^argument (&{ paramError l c /= Just paramFormMessage }) ^paramform (&{ paramError l c == Nothing }) ^paramtype (&[,)]) ^argcode { param l c }

# 選択肢と式 -----------------------------------------------------------------------
Alts :: [Alt]
Alts <- a:Alt as:MoreAlt* { a : as }

MoreAlt :: Alt
MoreAlt <- Ws '/' Alt

Alt :: Alt
Alt <- Ws l:Here is:Item(0)* Ws act:Action? c:Constructor? { Alt l (map fst is) act c }

# 構築子の名前（ADR 0075）。アクションを書く位置に置く
Constructor :: Located Text
Constructor <- Ws '@' l:Here n:(Ident ^constructor) { Located l n }

# 式の入れ子の深さは 256 段まで（ADR 0080）。
# 式の規則は、外側の群、前置の演算子、雛形の引数の段数を引数 d に受け取り、
# 式とその内側の段数の組を返す。後置の演算子とラベルは式を読んだ後に現れるので、
# 内側の段数を足して数える。上限を超える段に入る手前で ^nest を送出するので、
# 深い入れ子の解析は上限の段で止まる
Item(d :: Int) :: (Item, Int)
Item <- Ws !DeclStart l:Here b:Binding e:(PrefixE(d) ^expression) { (Item l (Just b) (fst e), snd e) }
     / Ws !DeclStart l:Here e:PrefixE(d) { (Item l Nothing (fst e), snd e) }

DeclStart <- Ident TemplateParams? HsParen? Ws ('::' / '<-') { () }

Binding :: Text
Binding <- Ident ':' !':'

# 先読みの選択肢を意味述語の選択肢より先に試す。
# 意味述語の選択肢が { の位置で失敗すると、深さの誤りがその位置で報告されるためである
PrefixE(d :: Int) :: (Expr, Int)
PrefixE <- Ws &'&' !('&' Ws '{') (&{ d < maxDepth }) ^nest '&' Ws e:(PrefixE(d + 1) ^expression) { deeper EAnd e }
        / Ws '&' Ws '{' c:PredCode { (EPred True c, 0) }
        / Ws &'!' !('!' Ws '{') (&{ d < maxDepth }) ^nest '!' Ws e:(PrefixE(d + 1) ^expression) { deeper ENot e }
        / Ws '!' Ws '{' c:PredCode { (EPred False c, 0) }
        / Ws &'$' (&{ d < maxDepth }) ^nest '$' e:(PrefixE(d + 1) ^expression) { deeper ECapture e }
        / Labeled(d)

Labeled(d :: Int) :: (Expr, Int)
Labeled <- e:SuffixE(d) lab:LabelPart(d + snd e)? { maybe e (\(l, n) -> deeper (\x -> ELabel l x n) e) lab }

LabelPart(d :: Int) :: (Loc, Text)
LabelPart <- Ws l:Here &'^' (&{ d < maxDepth }) ^nest '^' n:Ident { (l, n) }

SuffixE(d :: Int) :: (Expr, Int)
SuffixE <- p:Primary(d) ops:PostOps(d + snd p) { (foldl applyOp (fst p) ops, snd p + length ops) }

# 後置の演算子の列。一つごとに一段深くなる
PostOps(d :: Int) :: [Char]
PostOps <- c:PostOp(d) cs:PostOps(d + 1) { c : cs }
        / { [] }

PostOp(d :: Int) :: Char
PostOp <- Ws &[*+?] (&{ d < maxDepth }) ^nest c:$[*+?] { firstChar c }

Primary(d :: Int) :: (Expr, Int)
Primary <- Ws o:$Lit1 { (ELit (decodeLiteral o) o, 0) }
        / Ws o:$Lit2 { (ELit (decodeLiteral o) o, 0) }
        / Ws o:$ClassLit { (EClass (classRanges o) o, 0) }
        / Ws '.' { (EAny, 0) }
        / Ws &'(' (&{ d < maxDepth }) ^nest '(' g:GroupAlts(d + 1) Ws (!'{') ^groupaction (')' ^paren) { deeper EGroup g }
        / Ws l:Here n:Ident args:CallArgs { (ERef l n args, 0) }
        / Ws l:Here n:Ident as:TemplateArgs(d) vs:CallArgs? { deeper (\x -> ETemplate l n x (maybe [] id vs)) as }
        / Ws l:Here n:Ident !'(' { (ERef l n [], 0) }

# 雛形の呼び出しの引数。引数は選択と逐次を含む式であり、, か > で終わる。
# > の直後に空白なしで ( が続けば、値の引数である（ADR 0055）
TemplateArgs(d :: Int) :: ([Expr], Int)
TemplateArgs <- &('<' !'-') (&{ d < maxDepth }) ^nest '<' a:TemplateArg(d + 1) as:(Ws ',' TemplateArg(d + 1))* Ws '>' { (map fst (a : as), maximum (map snd (a : as))) }

TemplateArg(d :: Int) :: (Expr, Int)
TemplateArg <- g:GroupAlts(d) { case fst g of { [[Item _ Nothing e]] -> (e, snd g); gs -> (EGroup gs, snd g) } }

CallArgs :: [Located Text]
CallArgs <- '(' ArgChunks

GroupAlts(d :: Int) :: ([[Item]], Int)
GroupAlts <- s:Item(d)* ss:MoreGroup(d)* { withDepth (s : ss) }

MoreGroup(d :: Int) :: [(Item, Int)]
MoreGroup <- Ws '/' Item(d)*

# リテラルと文字クラス -------------------------------------------------------------------
Lit1 <- '\'' (Escape / !['\\\n] .)* ('\'' ^literal)

Lit2 <- '"' (Escape / !["\\\n] .)* ('"' ^literal)

Escape <- '\\' (EscapeBody ^escape)

EscapeBody <- [ntr\\'"\]\[\-]
           / 'u' ('{' ^ubrace) h:$[0-9a-fA-F]* ('}' ^uclose) (&{ validHex h }) ^uescape { () }

ClassLit <- '[' (!']') ^emptyclass ClassItem* (']' ^class)

ClassItem <- s:$(ClassChar ('-' !']' ClassChar)?) (&{ nonEmptyRanges s }) ^range { () }

ClassChar <- Escape
          / ![\]\\\n] .

# Haskell の断片 -------------------------------------------------------------------
Action :: Located Text
Action <- '{' l:Here c:$ActionCode ('}' ^action) { Located l c }

PredCode :: Located Text
PredCode <- l:Here c:$ActionCode ('}' ^action) { Located l c }

ArgChunks :: [Located Text]
ArgChunks <- c:ArgChunk ',' rest:ArgChunks { c : rest }
          / c:ArgChunk ')' { [c] }

ArgChunk :: Located Text
ArgChunk <- Ws l:Here c:$ArgCode (&{ not (T.null (T.strip c)) }) ^argument (&[,)]) ^argcode { Located l c }

# 深さ 0 の '}' の手前まで
ActionCode <- HsPart*

# 深さ 0 の ',' か ')' の手前まで
ArgCode <- (!',' HsPart)*

HsPart <- HsNested
       / HsString
       / HsChar
       / HsIdent
       / ![()\[\]{}"'] .

HsParen <- '(' HsInner* ')'

HsNested <- HsParen
         / '[' HsInner* ']'
         / '{' HsInner* '}'

HsInner <- HsNested
        / HsString
        / HsChar
        / HsIdent
        / ![()\[\]{}"'] .

HsString <- '"' ('\\' . / !'"' .)* ('"' ^hsstring)

# 識別子の文字の直後にない引用符は文字リテラルの始まり。
# 引用符はこの規則と HsString でだけ読む。閉じていない文字列や文字リテラルは、
# ラベルがなくても拒否される（ラベルは誤りの文言のため、ADR 0028）
HsChar <- '\'' '\\' . (!'\'' .)* ('\'' ^hschar)
       / '\'' . '\''
       / '\'' !'\\'

HsIdent <- c:$. &{ isIdentChar (firstChar c) && firstChar c /= '\'' } IdentChar* { () }
