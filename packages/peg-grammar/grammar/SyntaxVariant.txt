# 回復の規則（ADR 0035）。どのラベルの送出からも、次の最上位の要素の手前まで
# 読み飛ばす。行頭の空白、/、# でない文字を、最上位の要素の始まりとみなす
SkipToTop :: ()
SkipToTop <- (!('\n' ![ \t\r\n/#]) .)* { () }

RecoverTop :: Top
RecoverTop <- SkipToTop { TVerbatim (Loc 0 0) T.empty }

RecoverText :: Text
RecoverText <- SkipToTop { T.empty }

RecoverExpr :: Expr
RecoverExpr <- SkipToTop { EAny }
