# ADR 0063：型が () の規則で、値のない選択肢のアクションを省略する
# ../fixtures/UnitActions.hpeg の、省略したアクションを { () } と書いた版
# 省略できる { () } を残すので、拡張子を .txt にして peg-hs lint と peg-hs fmt の探索から外す
%module Fixture.UnitActionsExplicit

%{
import qualified Data.Text as T
}%

Start :: [T.Text]
Start <- Sp Item* !.

Item :: T.Text
Item <- Kw Name End
     / Op<Sep> Name End

Name :: T.Text
Name <- !Kw $[a-z]+

# リテラルと先読み
Kw :: ()
Kw <- 'if' ![a-z] Sp { () }
   / 'do' ![a-z] Sp { () }

# 単位の要素が二つ、束縛した単位の要素が二つと意味述語、単位の要素が一つ
End :: ()
End <- Sp ';' Sp { () }
    / a:Sp b:Sp &{ a == b } ',' { () }
    / Sp { () }

# 雛形の展開した規則の型が () になる
Op<p :: t> :: t
Op<p> <- '(' p ')'
      / '-' Sp { () }

# リテラルと意味述語
Sep :: ()
Sep <- '|' Sp { () }
    / '&' &{ True } { () }

# リテラルだけ
Sp :: ()
Sp <- ' '* { () }
